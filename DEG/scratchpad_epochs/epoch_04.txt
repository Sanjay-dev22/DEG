        ```js
            if (outcome === "nack") return '<span class="badge nack">400 · NACK</span>';
            if (outcome === "async") return '<span class="badge async">async</span>';
            return '<span class="badge ack">200 · ACK</span>';
          }

          // Renders the "full endpoint" reference line inside the Address block —
          // additive only, never replaces the existing From/To/Path/Action rows.
          // h.full = { method: "GET"|"POST"|"", url: "<real endpoint or internal note>" }
          function renderEndpointLine(full) {
            var m = (full.method || "").toUpperCase();
            var cls = m === "GET" ? "get" : (m === "POST" ? "post" : "none");
            var label = m || "—";
            return '<div class="endpoint-line"><span class="em ' + cls + '">' + label + '</span><span>' + full.url + "</span></div>";
          }
        ```
